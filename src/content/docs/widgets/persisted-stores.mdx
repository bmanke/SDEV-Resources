## How it works

```ts title="src/lib/stores/persisted.ts"
export function persisted<T>(key: string, initial: T) {
	const store = writable<T>(read(key, initial), (set) => {
		// Sync across tabs
		const onStorage = (e: StorageEvent) => e.key === key && set(JSON.parse(e.newValue!));
		addEventListener('storage', onStorage);
		return () => removeEventListener('storage', onStorage);
	});
	return {
		...store,
		set(value: T) {
			localStorage.setItem(key, JSON.stringify(value));
			store.set(value);
		},
	};
}

export const packageManager = persisted('docs:package-manager', 'npm');
```

In a component, `$packageManager` subscribes automatically:

```svelte
<script lang="ts">
	import { packageManager } from '../../lib/stores/persisted';
	const command = $derived(`${$packageManager} add ${pkg}`);
</script>
```

<Aside type="note">
	The store is SSR-safe. On the server it returns the initial value, and on the client it reads from `localStorage` when the island hydrates.
</Aside>
